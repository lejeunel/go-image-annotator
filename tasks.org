#+begin_src 
openapi: 3.1.0

info:
title: Image API
version: 1.0.0

paths:
/api/images:
    post:
    summary: Ingest an image
    operationId: ingestImage

    requestBody:
        required: true
        content:
        multipart/form-data:
            schema:
            type: object
            required:
                - metadata
                - image
            properties:
                metadata:
                $ref: '#/components/schemas/ImageMetadata'

                image:
                type: string
                format: binary
                description: Raw image data.

            encoding:
            metadata:
                contentType: application/json

            image:
                contentType: image/*

    responses:
        '201':
        description: Image successfully ingested
        content:
            application/json:
            schema:
                $ref: '#/components/schemas/ImageResponse'

        '400':
        description: Invalid request

        '413':
        description: Image too large

        '415':
        description: Unsupported image format

components:
schemas:

    ImageMetadata:
    type: object
    required:
        - filename
    properties:
        filename:
        type: string
        description: Original filename.
        description:
        type: string
        tags:
        type: array
        items:
            type: string

    ImageResponse:
    type: object
    required:
        - id
    properties:
        id:
        type: string
        description: Identifier of the ingested image.
#+end_src
